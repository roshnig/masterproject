# create a file inside src --> add a new file setupTests.ts

# create a vitest.config.ts file in route (optional, we can add this configuration in vite.config.ts as well if don't want
  to add sepearte file).
  Here you can define which files you want to include or exclude for tests.
  v8 coverage reporter as well
  
# in tsconfig.json and in tsconfig.app.json file, add compiler types-
  "compilerOptions": {
    "types": [
      "vitest/globals",
      "@testing-library/jest-dom"
    ],
  }
# add test script in package.json file
    "test": "vitest",
    "test:coverage": "vitest run --coverage"   //npm run test:coverage to see the coverage
    "test:ui": "vitest --ui"  //if already installed "@vitest/ui": "^3.2.4",
    
# If want to see test coverage in browser, then in file explorer , open your frontend project folder --> coverage -->
 click index.html. here you will see coverage report in browser
